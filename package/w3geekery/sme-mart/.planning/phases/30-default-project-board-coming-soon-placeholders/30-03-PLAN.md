```typescript
import { Component, inject } from '@angular/core';
import { ActivatedRoute } from '@angular/router';
import { toSignal } from '@angular/core/rxjs-interop';
import { map } from 'rxjs/operators';
import { ZbEmptyStateContainerComponent } from '@zerobias-org/ngx-library';

export interface ComingSoonData {
  title: string;
  description: string;
  featureKey: string;
}

@Component({
  selector: 'sme-mart-feature-coming-soon',
  standalone: true,
  imports: [ZbEmptyStateContainerComponent],
  templateUrl: './feature-coming-soon.component.html',
  styleUrls: ['./feature-coming-soon.component.scss'],
})
export class FeatureComingSoonComponent {
  private readonly route = inject(ActivatedRoute);

  // Reactive bridge from route.data → signal. toSignal cleans up automatically on destroy.
  readonly comingSoonData = toSignal(
    this.route.data.pipe(
      map((data) => ({
        title: (data['title'] as string) || 'Coming Soon',
        description: (data['description'] as string) || 'This feature is on the roadmap.',
        featureKey: (data['featureKey'] as string) || 'unknown',
      } as ComingSoonData))
    ),
    { initialValue: undefined }
  );
}
```

**Key points:**
- Component is lightweight (~25 lines) and fully standalone
- Uses ActivatedRoute + `toSignal` to bridge route.data → signal (no manual subscribe, no OnDestroy cleanup needed)
- Template reads `comingSoonData()` as a signal call; @if guards against the initial undefined window
- Imports ZbEmptyStateContainerComponent from ngx-library — NO CommonModule (Angular 21 built-in @if covers control flow)
- No API calls, no async operations beyond route observable
- Handles missing route data gracefully with fallback labels

**Anti-patterns to avoid:**
- Do NOT add CommonModule — Angular 21 `@if`/`@for` are built-in and CommonModule's only legitimate need here would be `*ngIf`/`*ngFor`, which are banned by modernization rules
- Do NOT declare unused signal `input()` declarations — route.data is the source of truth; signal inputs are dead code for a routed component with no parent
- Do NOT use `route.data.subscribe(...)` writing to a plain field — that's the legacy pattern; `toSignal()` is the Angular 21 equivalent and handles cleanup
- Do NOT use @Input/@Output decorators — this is a routed component with no parent/child binding
- Do NOT call any backend APIs or services (no EngagementsService, no SDK calls)
- Do NOT use *ngIf/*ngFor in the template — use @if/@for control-flow directives
