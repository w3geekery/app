```typescript
import { Component, OnInit, inject, input, computed } from '@angular/core';
import { ActivatedRoute } from '@angular/router';
import { CommonModule } from '@angular/common';
import { ZbEmptyStateContainerComponent } from '@zerobias-org/ngx-library';

export interface ComingSoonData {
  title: string;
  description: string;
  featureKey: string;
}

@Component({
  selector: 'sme-mart-feature-coming-soon',
  standalone: true,
  imports: [CommonModule, ZbEmptyStateContainerComponent],
  templateUrl: './feature-coming-soon.component.html',
  styleUrls: ['./feature-coming-soon.component.scss'],
})
export class FeatureComingSoonComponent implements OnInit {
  private readonly route = inject(ActivatedRoute);

  // Data from route.data: { title, description, featureKey }
  title = input<string>('');
  description = input<string>('');
  featureKey = input<string>('');

  // Read from route on init (route data doesn't auto-flow to inputs)
  comingSoonData?: ComingSoonData;

  ngOnInit() {
    // Extract coming-soon metadata from route.data
    this.route.data.subscribe((data) => {
      this.comingSoonData = {
        title: data['title'] || 'Coming Soon',
        description: data['description'] || 'This feature is on the roadmap.',
        featureKey: data['featureKey'] || 'unknown',
      };
    });
  }
}
```

**Key points:**
- Component is lightweight (~40 lines) and fully standalone
- Uses ActivatedRoute to read route.data (no props from parent)
- Inputs are declared but route.data subscription takes precedence (route data is async, inputs are sync)
- Imports ZbEmptyStateContainerComponent from ngx-library
- No API calls, no async operations beyond route subscription
- Handles missing data gracefully with fallback labels

**Anti-patterns to avoid:**
- Do NOT use @Input/@Output decorators — use input()/output() signals
- Do NOT add CommonModule alongside ZbEmptyStateContainerComponent imports — ZbEmptyStateContainerComponent has its own imports
- Do NOT call any backend APIs or services (no EngagementsService, no SDK calls)
- Do NOT add changeDetection: ChangeDetectionStrategy.OnPush — not needed for this simple component
- Do NOT use *ngIf/*ngFor — use @if/@for control-flow directives in the template
